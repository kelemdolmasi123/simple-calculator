// Simple calculator behavior
const display = document.getElementById('display');
const keys = document.querySelector('.keys');

let current = ''; // current expression shown
let resetNext = false; // whether next number should reset current

function updateDisplay(text){
display.textContent = text || '0';
}

function appendValue(val){
if (resetNext){
current = '';
resetNext = false;
}
// avoid multiple leading zeros
if (current === '0' && val === '0') return;
// avoid two decimals in a row in a number segment
if (val === '.') {
const parts = current.split(/[\+\-\*\/]/);
if (parts[parts.length-1].includes('.')) return;
}
current += val;
updateDisplay(current);
}

function clearAll(){
current = '';
updateDisplay('0');
}

function deleteLast(){
if (current.length > 0){
current = current.slice(0, -1);
updateDisplay(current || '0');
}
}

function percent(){
if (!current) return;
try {
const num = evaluateExpression(current);
const result = String(num / 100);
current = result;
updateDisplay(result);
resetNext = true;
} catch(e) { updateDisplay('Error') }
}

function evaluateExpression(expr){
// Replace unicode operators for eval
expr = expr.replace(/×/g, '*').replace(/÷/g, '/').replace(/−/g, '-');
// Allow only digits, operators, parentheses, dot and spaces
if (!/^[0-9+\-*/().\s]+$/.test(expr)) throw new Error('Invalid characters');
// eslint-disable-next-line no-new-func
const fn = new Function('return ' + expr);
const result = fn();
if (!isFinite(result)) throw new Error('Math error');
// limit floating precision
return Math.round((result + Number.EPSILON) * 1e12) / 1e12;
}

function calculate(){
if (!current) return;
try {
const result = evaluateExpression(current);
updateDisplay(String(result));
current = String(result);
resetNext = true;
} catch (e) {
updateDisplay('Error');
resetNext = true;
}
}

keys.addEventListener('click', e => {
const btn = e.target.closest('button');
if (!btn) return;
const val = btn.dataset.value;
const action = btn.dataset.action;

if (action === 'clear') clearAll();
else if (action === 'delete') deleteLast();
else if (action === 'percent') percent();
else if (action === 'calculate') calculate();
else if (val) appendValue(val);
});

// Keyboard support
window.addEventListener('keydown', e => {
if (e.key >= '0' && e.key <= '9') appendValue(e.key);
else if (e.key === '.') appendValue('.');
else if (['+','-','/','*','(',')'].includes(e.key)) appendValue(e.key);
else if (e.key === 'Enter') { e.preventDefault(); calculate(); }
else if (e.key === 'Backspace') deleteLast();
else if (e.key === 'Escape') clearAll();
else if (e.key === '%') percent();
});
